
get_property(WINMD_SOURCES GLOBAL PROPERTY WINMD_SOURCES)

set(WINMD_GROUPS "")
foreach(IDL ${WINMD_SOURCES})
    get_filename_component(NAME ${IDL} NAME_WLE)
    if(NAME MATCHES "^windows\\.ui\\.xaml(\\.|$)")
        set(GROUP windows.ui.xaml)
    elseif(NAME MATCHES "^([^.]+\\.[^.]+)")
        set(GROUP ${CMAKE_MATCH_1})
    else()
        set(GROUP ${NAME})
    endif()
    list(APPEND WINMD_GROUP_${GROUP} ${IDL})
    list(APPEND WINMD_GROUPS ${GROUP})
endforeach()
list(REMOVE_DUPLICATES WINMD_GROUPS)

set(WINMD_OUTPUTS "")
foreach(GROUP ${WINMD_GROUPS})
    set(CONTENT "")
    foreach(IDL ${WINMD_GROUP_${GROUP}})
        get_filename_component(IDL_NAME ${IDL} NAME)
        string(APPEND CONTENT "import \"${IDL_NAME}\";\n")
    endforeach()
    set(AGGREGATE ${CMAKE_CURRENT_BINARY_DIR}/${GROUP}-winmd.idl)
    file(GENERATE OUTPUT ${AGGREGATE} CONTENT "${CONTENT}")

    set(NAMESPACES ${GROUP})
    foreach(OTHER ${WINMD_GROUPS})
        string(FIND ${OTHER} "${GROUP}." POSITION)
        if(POSITION EQUAL 0)
            list(APPEND NAMESPACES "!${OTHER}")
        endif()
    endforeach()

    set(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/winmd/${GROUP}.winmd)
    add_winmd_file(${OUTPUT} ${AGGREGATE} NAMESPACES ${NAMESPACES} DEPENDS ${WINMD_GROUP_${GROUP}})
    list(APPEND WINMD_OUTPUTS ${OUTPUT})
endforeach()

add_custom_target(winmetadata DEPENDS ${WINMD_OUTPUTS})
foreach(OUTPUT ${WINMD_OUTPUTS})
    add_cd_file(TARGET winmetadata FILE ${OUTPUT} DESTINATION reactos/system32/WinMetadata FOR all)
endforeach()
